which free vmstat pmap ps top
which valgrind heaptrack
# Debian/Ubuntu: apt install procps sysstat valgrind
# RHEL/CentOS : yum install procps-ng sysstat valgrind

# 2. 看系统内存基线
free -m

# 3. 看内核的 overcommit 策略
cat /proc/sys/vm/overcommit_memory      # 0=启发式 1=总是允许 2=严格
cat /proc/sys/vm/overcommit_ratio       # 仅模式 2 生效，默认 50

# 4. 看透明大页（影响 fork 和内存碎片）
cat /sys/kernel/mm/transparent_hugepage/enabled

# 5. 看 cgroup 版本和当前限制

cat /sys/fs/cgroup/memory/memory.limit_in_bytes 2>/dev/null   # v1

# 6. 换页统计（有没有在用 swap）
cat /proc/vmstat | grep -E "pgpgin|pgpgout|pswpin|pswpout"
看内存数据前必须知道的两件事：
① free 的 "free" 列几乎总是很小，这是正常的。Linux 会主动把空闲内存拿去做页缓存（page cache），有需要时立刻回收。判断内存够不够看 available 列，不是 free 列。
② 容器里的 free 显示的是宿主机内存，不是你的配额。要看 /sys/fs/cgroup/memory.max 和 memory.current。
模块 1

虚拟内存解决了什么

虚拟内存不是"把硬盘当内存用"这种科普说法。它真正的价值是让每个进程拥有一份独立、连续、可以任意撒谎的地址空间，而物理内存的分配完全由内核在背后调度。

四个核心能力

能力机制带来的好处
隔离每个进程一份页表，虚拟地址到物理地址的映射互不相干进程 A 无法读写进程 B 的内存（除非显式共享）。越界访问只会崩自己，不会崩别人
超配
overcommit		分配虚拟地址时不立刻给物理页，等到真正访问才触发缺页并分配进程可以 malloc 10 GB 但只用 100 MB。这也是 OOM 的根源
简化链接与加载每个进程都以为自己从固定地址开始（如 x86-64 的 0x400000）可执行文件不用关心自己被加载到哪里；动态库可以共享同一份物理页
共享与写时复制多个进程映射同一物理页，写入时才复制（COW）fork 几乎零成本；动态库、只读数据只占一份物理内存

地址空间长什么样

# x86-64 Linux 进程的典型布局（低地址 → 高地址）
0x0000000000400000   代码段 .text（只读、可执行）
                     数据段 .data / .bss（可读写）
                     ↓
                     堆 heap  ← brk() 向上增长
                     ↓
                     （未映射的空洞，巨大）
                     ↓
                     内存映射区 mmap ← 向下增长（动态库、大块 malloc、文件映射）
                     ↓
0x00007ffffffff000   栈 stack ← 向下增长
                     ↓
0xffffffffffffffff   内核空间（用户态不可访问）

# 关键数字
用户空间可用范围：0x000000000000 ~ 0x00007fffffffffff = 128 TB
（4 级页表，48 位虚拟地址）

# 所以一个进程的 VSZ 可以轻松到几十 GB，而 RSS 只有几十 MB
# VSZ 大不代表占内存，只代表"地址空间用掉了"

overcommit 的三种模式

overcommit_memory行为适用
0（默认）启发式。内核按经验判断这次分配是否"明显过分"，过分就拒绝大多数场景。允许合理超配，但挡住明显离谱的请求
1总是允许。malloc 几乎不会失败，直到真的没内存时 OOM Killer 出手Redis 官方推荐（避免 fork 时因 COW 预估失败）；科学计算
2严格。承诺的虚拟内存总量不能超过 swap + 物理内存 × ratio需要确定性行为的场景。但会导致 mmap 大量失败
# 模式 2 的可用额度计算
CommitLimit = swap 总量 + 物理内存 × overcommit_ratio / 100
# 默认 ratio = 50，16 GB 内存 + 0 swap → CommitLimit = 8 GB
# 所有进程的已承诺虚拟内存之和不能超过这个值

# 查看当前承诺量
grep -E "CommitLimit|Committed_AS" /proc/meminfo

# 常见现象：模式 2 下 Java 启动失败
# "Native memory allocation (mmap) failed to map ..."
# 因为 JVM 预留的虚拟地址（堆 + metaspace + 保留区）超过了 CommitLimit
超配是 OOM 的根源，也是它的价值所在。
因为 malloc 成功不等于物理内存到手。一个进程可以顺利 malloc 8 GB，然后在逐页写入的过程中被 OOM Killer 杀掉。这不是 bug，是设计 —— 否则每个进程都得按最坏情况预留内存，机器利用率会低得可怕。
工程含义：不要用 "malloc 有没有返回 NULL" 来判断内存够不够。要看 RSS 和 cgroup 的 memory.current。
模块 2

页表与地址翻译

为什么是四级

# 48 位虚拟地址 + 4 KB 页（12 位偏移）
页内偏移 offset = 12 位
需要翻译的位   = 48 - 12 = 36 位

# 如果只用一级页表：2^36 个页表项 × 8 字节 = 2 TB
# 每个进程都要 2 TB 页表 → 完全不可行

# 分成四级，每级 9 位（512 项，正好一个 4 KB 页装 512 个 8 字节表项）
PGD  (Page Global Directory)   9 位  →  512 项
PUD  (Page Upper Directory)    9 位  →  512 项
PMD  (Page Middle Directory)   9 位  →  512 项
PTE  (Page Table Entry)        9 位  →  512 项
offset                        12 位
                              ─────
                              48 位

# 关键：按需分配。进程只用了几个 GB 的地址空间，
# 只需要为实际用到的区域建表，大部分层级是空的
# 一个典型进程的页表实际只占几 MB

# 5 级页表（Ice Lake 起支持）扩展到 57 位 = 128 PB
# 但会多一次内存访问，默认关闭

一次地址翻译的代价

四级页表意味着一次逻辑上的内存访问要读 4 次页表 + 1 次数据 = 5 次内存访问。如果每次都这样，性能会崩掉。救场的是 TLB。

结构容量说明
L1 dTLB64 项，4 路命中延迟 1 周期，几乎免费
L2 STLB1536~2048 项命中延迟 7~10 周期
页表遍历—4 次内存访问，可能 20~100 周期，还可能本身 cache miss
# 4 KB 页、TLB 2048 项，能覆盖多大范围？
2048 × 4 KB = 8 MB

# 也就是说：工作集超过 8 MB，TLB 就开始 miss
# 而现代服务的堆动辄几个 GB —— 所以大内存应用几乎必然 TLB 抖动

# 解法：大页（huge page）
2 MB 大页：2048 × 2 MB = 4 GB 覆盖范围（提升 512 倍）
1 GB 大页：2048 × 1 GB = 2 TB

# 代价：内部碎片。用 2 MB 页存 100 字节的数据，浪费 2 MB
# 所以大页只适合大块、连续、长期存在的内存（JVM 堆、数据库缓冲池）

缺页中断的两种类型

类型触发条件代价
Minor fault
次缺页		页已经在物理内存里，只是当前进程的页表里没有映射。典型场景：COW 写时复制、进程刚 fork、页被换出后换回但还在缓存约 1~2 μs，纯内存操作
Major fault
主缺页		页不在内存里，必须从磁盘/swap 读进来几十 μs 到几 ms，取决于存储介质
# 怎么看缺页次数
ps -o min_flt,maj_flt -p 1234
# min_flt = 次缺页累计，maj_flt = 主缺页累计

# 实时观察
pidstat -r -p 1234 1
#   minflt/s  majflt/s     VSZ     RSS   %MEM  Command
#    1200.00      0.00  8123456  456789   5.6  java

# 系统级
vmstat 1
# si/so 列 = 每秒从 swap 换入/换出的 KB 数
# si/so 持续非 0 → 内存压力大，正在用 swap，性能会断崖式下跌

# 详细统计
cat /proc/vmstat | grep -E "^pgfault|^pgmajfault"
# pgfault     = 总缺页数（含 minor）
# pgmajfault  = 主缺页数
次缺页不是免费的。一次 minor fault 要进内核、分配页、填页表、返回，约 1~2 μs。如果一个程序每秒产生 10 万次 minor fault，就是 0.2 秒/秒 —— 20% 的 CPU 花在缺页上。
典型来源：① 频繁 malloc/free 大块内存（每次都 mmap/munmap）；② JVM 堆的首次触碰（-XX:+AlwaysPreTouch 可以在启动时预触碰，把成本前置）；③ fork 之后大量写（COW）。

页表项里有什么

# 一个 64 位 PTE 的典型位布局（x86-64）
bit 0     Present   # 页是否在内存中。为 0 时其余位由内核自由使用
bit 1     R/W       # 0=只读 1=可写
bit 2     User      # 1=用户态可访问 0=仅内核
bit 3     PWT       # 写穿透（几乎不用）
bit 4     PCD       # 禁用缓存（映射设备内存时用）
bit 5     Accessed  # 被访问过。内核靠它做页面置换（LRU 近似）
bit 6     Dirty     # 被写过。决定换出时要不要写回磁盘
bit 7     PAT / PS  # 大页标志（PS=1 时这一级就是最终映射）
bit 8     Global    # 全局页，切换进程时不刷 TLB（内核页用）
bit 9-11  软件保留  # Linux 用来存 swap 类型、COW 标记等
bit 12-51 物理页帧号 PFN
bit 63    NX        # 不可执行。W^X 保护的基础

# 注意 Accessed 和 Dirty 是硬件自动置位的
# 内核的 LRU 链表和脏页回写，全靠这两位驱动

堆分配器的三条路线

分配器设计特点
ptmalloc
glibc 默认		主 arena + 每线程 arena（64 位下最多 8 × 核数）。每个 arena 用 bins 管理空闲块通用性好，但多线程下锁竞争和碎片较严重。大块走 mmap
jemalloc
FreeBSD/Facebook		按 size class 分级，每线程 tcache，后台线程按 decay 回收脏页碎片控制好，多线程扩展性好。Redis 官方推荐
tcmalloc
Google		每线程 cache + 中央空闲链表 + page heap小对象分配极快，大对象走页级分配
# 换分配器：LD_PRELOAD 即可，不用改代码
LD_PRELOAD=/usr/lib/x86_64-linux-gnu/libjemalloc.so.2 ./app

# 验证是否生效
lsof -p <pid> | grep -i jemalloc
# 或者看 /proc/<pid>/maps

# glibc 的两个关键阈值（可以用 mallopt 调）
M_MMAP_THRESHOLD   默认 128 KB   # 超过就改用 mmap，free 时直接 munmap 还给系统
M_TRIM_THRESHOLD   默认 128 KB   # 堆顶空闲超过这个值就 brk 收缩

# 注意：M_MMAP_THRESHOLD 是动态的
# 每次 free 一个 mmap 块，阈值会上调，最高到 32 MB
# 目的是减少 mmap/munmap 的系统调用次数
# 副作用：程序跑久了之后，大块内存不再归还给系统

brk vs mmap

方式机制适用与代价
brk移动堆顶指针，扩展/收缩连续的一段小块、频繁分配。问题：只能从顶部收缩，中间的空闲块无法归还，容易形成"堆顶卡住"
mmap建立一段独立的匿名映射，free 时 munmap 直接归还大块（> 128 KB）、长期存活。代价：每次 mmap/munmap 都要改页表、触发 TLB 失效
# 这就是"内存只涨不降"的第一个原因
程序 malloc 了 1000 个 1 MB 的块（走 mmap）
free 掉其中 999 个 → 这些块 munmap 了，RSS 立刻下降
但如果它们是 100 KB 的块（走 brk）
free 后只是回到空闲链表，堆顶没降，RSS 不会下降

# 缓解手段
malloc_trim(0);                    # glibc 提供，主动收缩堆
mallopt(M_TRIM_THRESHOLD, 64*1024); # 降低收缩阈值
mallopt(M_MMAP_THRESHOLD, 64*1024); # 让更多分配走 mmap，便于归还

# 但注意：malloc_trim 会遍历所有 arena，本身有开销
# 不要在高频路径调用
模块 4

内存泄漏与内存溢出

这两个词经常被混用，但它们的成因、表现、排查手段完全不同。

维度内存泄漏 Leak内存溢出 Overflow / OOM
定义分配了内存但永远不再释放，且失去了引用申请的超过可用上限，或者写入越过缓冲区边界
表现内存缓慢、持续上涨，几天到几周后 OOM突然失败：malloc 返回 NULL、进程被 kill、或崩溃
时间尺度慢，通常要观察趋势才能发现快，往往是某个瞬间的峰值
典型原因忘记 free/delete、异常路径漏释放、循环引用、缓存无上限、监听器未注销一次性加载大文件、批量查询没分页、缓冲区越界、递归爆栈
排查工具valgrind、heaptrack、pprof、堆 dumppmap、cgroup memory.events、dmesg
注意第三个含义："缓冲区溢出"（buffer overflow）是另一回事 —— 写越界，属于安全漏洞（CWE-787）。它和内存不够无关，但中文里也常被叫"内存溢出"。看到这个词要先问清楚指哪个。

泄漏的五种常见成因

异常路径漏释放。C++ 里 new 之后抛异常，后面的 delete 没执行。解法：用 RAII / 智能指针，让析构自动发生。
循环引用。A 持有 B，B 持有 A，引用计数永远不归零。Java/Python 的 GC 能处理，但 shared_ptr 不能。解法：一端用 weak_ptr。
缓存没有上限。用 map 做本地缓存，只 put 不 evict。解法：用带容量限制的 LRU（Caffeine、Guava Cache、container/list 手写）。
监听器/回调没注销。注册了事件监听但对象销毁时没移除，容器一直持有引用。解法：成对的生命周期管理，或用弱引用。
线程局部变量。ThreadLocal 在线程池里不会自动清理，线程复用导致旧值一直挂着。解法：用完必须 remove()。

怎么定位

# ===== C/C++：valgrind（慢但准，约 20~50 倍开销）=====
valgrind --leak-check=full --show-leak-kinds=all \
         --track-origins=yes ./app

# 输出里重点看三段
definitely lost:  # 确定泄漏，没有指针指向它 —— 一定要修
indirectly lost:  # 因为别的块泄漏而跟着泄漏
possibly lost:    # 可能有指针指向块中间，需要人工判断
still reachable:  # 程序结束时还有指针，通常不算泄漏（比如全局单例）

# ===== heaptrack：采样式，开销小得多，适合长跑 =====
heaptrack ./app
heaptrack_gui heaptrack.app.*.zst

# ===== 生产环境：pmap 看趋势 =====
pmap -x <pid> | tail -1        # 看 RSS 总量
# 每隔 10 分钟采一次，画成曲线。持续上涨且不回落 = 泄漏

# ===== Java =====
jmap -histo:live <pid> | head -20      # 各类对象数量与占用
jmap -dump:live,format=b,file=heap.hprof <pid>
# 然后用 Eclipse MAT 或 VisualVM 分析支配树

# ===== Go =====
curl http://localhost:6060/debug/pprof/heap?debug=1
go tool pprof -http=:8080 http://localhost:6060/debug/pprof/heap
判断"是不是泄漏"的正确方法：看趋势，不看绝对值。
① 锯齿形（涨上去又掉下来）：正常，是 GC 或请求峰值，不是泄漏。
② 阶梯形（涨上去不降，过一段时间再涨一截）：典型泄漏，每次触发某条路径就漏一点。
③ 线性上涨：可能是缓存无上限，也可能是连接/文件描述符泄漏。
④ 稳定在一个高位：可能只是分配器不归还（模块 6 讲的碎片问题），不是泄漏。
观察窗口至少覆盖一个完整的业务周期（比如 24 小时），否则很容易误判。
模块 5

GC 原理

GC 解决的是"怎么在不靠人手动 free 的前提下，找出哪些内存没人用了"。所有算法都在吞吐、延迟、内存开销三者之间做取舍。

三种基础算法

算法做法问题
引用计数每个对象记一个被引用次数，归零就回收无法处理循环引用；每次赋值都要改计数，开销分散在业务代码里
标记-清除
mark-sweep		从根（栈、全局变量、寄存器）出发遍历可达对象，标记；未标记的回收产生外部碎片；标记阶段要暂停业务（STW）
标记-整理 / 复制标记后把存活对象挪到一起，消除碎片要移动对象，所有引用都得改；需要额外空间

分代假设

GC 领域最重要的经验规律：绝大多数对象都是朝生夕死的（弱分代假说）。所以：

区域存放回收策略
新生代 Young刚创建的对象频繁、快速回收（Minor GC）。用复制算法，因为存活对象少，复制成本低
老年代 Old熬过多次 GC 的对象不频繁、较慢（Major / Full GC）。用标记-清除或标记-整理

这个假设的价值在于：只扫描新生代就能回收 90% 的垃圾，成本远低于每次都扫全堆。

三色标记法

# 把对象分成三种颜色
白色：还没被访问到。GC 结束时仍是白色 = 垃圾，回收
灰色：自己访问到了，但它引用的对象还没扫完。待处理队列
黑色：自己和所有引用都扫完了。确定存活

# 流程
1. 所有对象初始为白色，根对象置灰
2. 从灰色队列取一个对象，把它引用的白色对象置灰，自己变黑
3. 重复直到灰色队列为空
4. 剩下的白色对象就是垃圾

# 问题：并发标记时，业务线程还在改引用
# 如果黑色对象新引用了白色对象，而这个白色对象
# 已经不会再被扫描到 —— 就会被错误回收（对象消失）

# 两种破坏条件（必须同时满足才会出错）
1. 黑色对象指向了白色对象（插入了一条新边）
2. 从灰色对象到该白色对象的所有路径都被删除了

# 所以只要破坏其中任意一个，就安全了

写屏障：两种流派

方式破坏哪个条件代表
增量更新
incremental update		破坏条件 1：黑色对象新增引用时，把它重新变灰，之后重新扫描CMS 的初始实现
SATB
Snapshot-At-The-Beginning		破坏条件 2：删除引用时，把被删除的旧值记录下来，按"GC 开始时的快照"来标记G1、ZGC 的部分阶段
混合写屏障插入和删除都记录，兼顾两者Go（1.8 起的 hybrid write barrier）

Java G1 与 Go 的差异

维度Java G1Go（1.5+）
堆布局切成最多 2048 个 Region（1~32 MB），逻辑上分 Eden/Survivor/Old/Humongous没有分代（Go 1.20 前），整堆统一管理
回收单位Region。按"回收价值"排序，优先收垃圾最多的 Region（Garbage First）整堆并发标记 + 清扫

写屏障SATB（标记阶段）+ 卡表（跨代引用）混合写屏障（插入 + 删除）

GOGC=100          # 默认。堆涨到上次存活量的 2 倍时触发 GC
                  # 调大 → GC 更少、CPU 更省、内存更高
                  # 调小 → 内存更省、GC 更频繁
GOGC=off          # 关闭 GC（只在极短生命周期程序里用）

GOMEMLIMIT=3GiB   # 1.19+。软限制，接近时会更激进地 GC
                  # 关键：它是软限制，不保证不超。真正的硬限制还是 cgroup

# 两者配合的正确姿势
GOMEMLIMIT = cgroup 内存上限 × 0.9
GOGC       = 保持默认或适度调大

# 查看 GC 行为
GODEBUG=gctrace=1 ./app
# 每行输出：gc 次数 @ 时间 s，堆大小 → 目标，STW 时长等
GC 的三个常见误解：
① "有 GC 就不会内存泄漏"。错。GC 只回收不可达的对象。缓存无上限、集合只增不减、ThreadLocal 不清理，都会导致"逻辑泄漏" —— 对象一直可达，GC 无能为力。
② "GC 语言不需要关心内存"。错。对象分配速率、对象大小、存活时间，直接决定 GC 频率和暂停时长。分配越少，GC 越轻松。
③ "STW 越短越好"。不完全对。压低 STW 通常要用更多并发标记，占用更多 CPU 和内存带宽，总吞吐会下降。这是取舍，不是免费的。
模块 6

内存碎片

"服务跑了三天，内存只涨不降" —— 大部分情况不是泄漏，是碎片。

类型含义例子
内部碎片
internal		分配出去的内存比实际需要的大，多出来的部分浪费在块内部申请 100 字节，分配器按 112 字节的 size class 给。多出的 12 字节谁也拿不到
外部碎片
external		空闲内存总量够，但分散成小块，凑不出连续的大块堆里有 100 个 1 MB 的空闲块，但申请 10 MB 连续内存失败

为什么 RSS 只涨不降

分配器按 size class 取整。jemalloc/tcmalloc 把请求映射到最近的 size class（如 96、112、128）。请求 100 字节实际占 112，长期积累就是 10%+ 的浪费。
free 只是还给分配器，不是还给内核。释放的内存进入空闲链表，等待下次复用。只有满足特定条件（brk 收缩、munmap、分配器的 decay 机制）才真正归还。
brk 只能从顶部收缩。如果堆顶恰好有个长期存活的小对象，下面再大的空闲块也归还不了。这就是"堆顶卡住"。
线程缓存不共享。线程 A 释放的内存进了 A 的 tcache，线程 B 要用时拿不到，只能向系统申请新的。线程数越多，碎片越严重。
分配模式随时间变化。启动时大量小对象，运行期变成大对象，原来的小块拼不出大块，只能不断向系统要新页。

怎么缓解

手段适用代价
换分配器jemalloc / tcmalloc 替代 ptmalloc需要 LD_PRELOAD 或重新编译；不同分配器的峰值内存可能不同
对象池 / 内存池高频创建销毁的同类型对象要自己管理生命周期，池太大会浪费
固定大小分配能预估大小的场景（如固定长度缓冲区）灵活性下降
jemalloc 的 decay长时间运行的服务dirty_decay_ms 调小会更积极归还，但增加系统调用
定期重启实在没办法时的兜底有损，且掩盖了真问题
# jemalloc 的关键参数（环境变量）
MALLOC_CONF="dirty_decay_ms:1000,muzzy_decay_ms:0,background_thread:true"

# dirty_decay_ms：脏页多久后归还给系统，默认 10000（10 秒）
# muzzy_decay_ms：madvise(MADV_FREE) 的页多久后真正释放，默认 0
# background_thread：用后台线程做回收，不阻塞业务线程

# 验证碎片程度
# 对 glibc：比较 RSS 和 malloc_usable_size 之和
# 对 jemalloc：看 stats.allocated vs stats.resident
MALLOC_CONF="stats_print:true" ./app

# 判断是不是碎片而非泄漏的关键实验
# 让服务空跑一段时间（不接流量），观察 RSS 是否下降
# 下降 → 是缓存/可回收，不是泄漏
# 不降 → 可能是碎片（分配器不归还），也可能是真泄漏
# 再触发一次完整 GC（jcmd GC.run / runtime.GC()）
# 仍不降 → 大概率是分配器层面的碎片
一个实用的判断流程：① 先确认是"持续上涨"还是"涨到某个水位就平稳"；② 触发 Full GC，看是否回落 —— 回落说明是 GC 没来得及收，不回落说明是可达对象（泄漏）或分配器不归还（碎片）；③ 用 pmap -x 看是哪一段在涨（heap / anon mmap / 动态库）；④ 对 anon 段持续增长且 GC 无效的，基本就是碎片或原生内存泄漏（DirectByteBuffer、JNI、Netty 的堆外内存）。
模块 7

OOM 与 cgroup

内核 OOM Killer 怎么选人

# 核心是 badness 打分，分数越高越可能被杀
badness ≈ 进程的 RSS + swap 使用量 + 页表占用

# 但这不是简单比大小，实际计算（简化）
oom_score 归一化到 0~1000
考虑因素：
  1. 物理内存占用（RSS）—— 主要权重
  2. swap 占用
  3. 页表大小
  4. oom_score_adj 调整值（-1000 ~ 1000）
  5. 是否 CAP_SYS_ADMIN 等特权（会减分）

# oom_score_adj 是人工干预的旋钮
-1000  →  永不被杀（比如 sshd、systemd）
    0  →  默认
 1000  →  优先被杀

# 查看和设置
cat /proc/<pid>/oom_score
cat /proc/<pid>/oom_score_adj
echo -500 > /proc/<pid>/oom_score_adj
OOM Killer 的"直觉"经常是错的：它倾向于杀占用内存最多的进程，而不是"最该负责的那个"。所以一个 8 GB 的数据库和一个 100 MB 的泄漏服务同时存在时，数据库会被先杀。
对策：给关键进程设 oom_score_adj = -1000（需要权限），或者用 systemd 的 OOMScoreAdjust=。但注意 —— 这只是转移了受害者，没解决内存不够的问题。

cgroup 的内存限制

文件（cgroup v2）含义
memory.max硬限制。超过就触发 cgroup 内 OOM，杀进程
memory.high软限制。超过会被限流（throttle），强制回收，但不杀进程
memory.current当前用量（含页缓存）
memory.stat明细：anon、file、slab、kernel_stack 等
memory.eventsoom / oom_kill 计数。判断"是不是被 OOM 杀了"的关键证据
memory.pressurePSI 指标，反映内存压力导致的停顿时间
# 判断容器是否发生过 OOM
cat /sys/fs/cgroup/memory.events
# low 0
# high 0
# max 12
# oom 3          ← 触发了 3 次 OOM
# oom_kill 3     ← 杀了 3 个进程

# 进程被杀的证据
dmesg -T | grep -i "killed process"
# [Wed Oct 15 10:23:41 2026] Out of memory: Killed process 12345 (java)
#   total-vm:8123456kB, anon-rss:3987654kB, file-rss:0kB

# 容器退出码 137 = 128 + 9 (SIGKILL) —— OOM 的典型信号
docker inspect --format='{{.State.OOMKilled}}' <container>
# true 就实锤了

# 注意：memory.current 包含页缓存
# 一个只读文件的容器，页缓存可能占满配额，看起来"内存满了"
# 实际上页缓存是可回收的，内核会先回收再 OOM
cat /sys/fs/cgroup/memory.stat | grep -E "^anon|^file"

为什么容器里的 JVM 要用 MaxRAMPercentage

# ===== 错误做法：写死 -Xmx =====
java -Xmx4g -jar app.jar
# 问题 1：容器限制改成 2 GB，-Xmx 还是 4 GB → 启动就 OOMKilled
# 问题 2：容器限制改成 8 GB，-Xmx 还是 4 GB → 一半内存浪费

# ===== 更错的做法：-Xmx 等于容器上限 =====
java -Xmx4g ...  # 容器 memory.max = 4g
# 堆之外还有很多内存开销：
#   Metaspace         几百 MB（取决于类数量）
#   Code Cache        默认 240 MB
#   线程栈            1 MB × 线程数
#   Direct ByteBuffer 堆外内存（Netty 大量使用）
#   GC 自身的元数据    （G1 的卡表、记忆集）
#   JVM 自身代码和数据
# 结果：堆还没满，容器已经 OOM 了

# ===== 正确做法：按比例 =====
java -XX:MaxRAMPercentage=75.0 \
     -XX:InitialRAMPercentage=75.0 \
     -XX:+UseContainerSupport \
     -jar app.jar

# UseContainerSupport 从 JDK 10 / 8u191 起默认开启
# 它会读 cgroup 的限制，把 "物理内存" 替换成 "容器可用内存"
# 然后 MaxRAMPercentage 按这个基数算堆大小

# 容器 4 GB，75% → 堆约 3 GB，留 1 GB 给堆外

# 验证实际生效的堆大小
java -XX:+PrintFlagsFinal -version | grep -i maxheap
jcmd <pid> VM.flags | tr ' ' '\n' | grep -i heap
参数默认值说明
MaxRAMPercentage25.0最大堆占容器内存的百分比。默认只有 25%，对服务型应用通常偏低
InitialRAMPercentage1.5625初始堆占比。设成和 Max 一样可以避免运行期扩容带来的抖动
UseContainerSupporttrueJDK 10+ / 8u191+ 默认开启
MaxDirectMemorySize= 最大堆堆外直接内存上限。Netty 应用要显式设置，否则可能吃掉和堆一样多的内存
MetaspaceSize约 21 MB 起无上限（受 MaxMetaspaceSize 限制），动态类加载多的应用要关注
一个真实的高频事故：容器限 2 GB，JVM 用 -Xmx2g，压测时被 OOMKilled，但 jmap 看堆只用了 1.2 GB。
原因：剩下的 800 MB 是堆外 —— Metaspace 200 MB + Code Cache 240 MB + 200 个线程栈 200 MB + Direct Buffer 150 MB。堆没满，但容器总量超了。
正确配置：-XX:MaxRAMPercentage=70（给堆外留 30%），并且显式设 -XX:MaxDirectMemorySize。
模块 8

观测：RSS、VSZ 与 available

free 的每一列到底是什么意思

# 典型输出
$ free -m
               total        used        free      shared  buff/cache   available
Mem:           15884        6210        1234         312        8439        9102
Swap:           2047          12        2035

# 逐列解释
total       = 物理内存总量
used        = total - free - buff/cache   # 已分配且不可回收
free        = 完全没被用到的内存，通常很小，这是正常的
shared      = tmpfs / 共享内存占用（会算进 used）
buff/cache  = 页缓存 + 块设备缓存。可回收
available   = 估算的"还能给新程序用多少"，这才是关键

# available 的估算方式（简化）
available ≈ free + 大部分可回收的页缓存 - 内核保留水位

# 为什么 free 很小是正常的
# Linux 的策略是"空闲内存就是浪费"，主动拿去做页缓存加速文件读写
# 有进程需要内存时，页缓存会被立刻回收
# 所以：看 available，不看 free

# 什么时候要警惕
available < total × 10%   → 内存紧张，随时可能 OOM
si/so 持续非 0            → 已经在用 swap，性能断崖

RSS 与 VSZ

指标全称含义
VSZVirtual Set Size进程虚拟地址空间的大小。包含已映射但从未访问的页、共享库、预留区域。几乎总是远大于实际占用
RSSResident Set Size进程实际占用物理内存的页数。包含共享页（同一个共享库在 10 个进程里各算一次）
PSSProportional Set Size共享页按共享进程数均摊。10 个进程共享 10 MB，每个算 1 MB。把所有进程的 PSS 加起来 ≈ 真实内存占用
USSUnique Set Size进程独占的物理内存。杀掉这个进程能释放多少 —— 判断泄漏最有用的指标
# 一个进程的四个数字
$ grep -E "VmSize|VmRSS|RssAnon|RssFile|RssShmem" /proc/1234/status
VmSize:   8123456 kB    # VSZ，约 8 GB
VmRSS:     456789 kB    # RSS，约 456 MB
RssAnon:   234567 kB    # 匿名页（堆、栈）—— 泄漏通常体现在这里
RssFile:   222222 kB    # 文件映射页（代码、mmap 的文件）
RssShmem:       0 kB    # 共享内存

# 更精确的 PSS / USS
cat /proc/1234/smaps_rollup
# Rss:     456789 kB
# Pss:     312456 kB   ← 共享部分已均摊
# Pss_Anon: 234567 kB
# Private_Clean / Private_Dirty: 独占部分
# Referenced: 最近被访问过的页

# pmap 看分段
pmap -x 1234 | tail -1
# 或者按大小排序看哪一段最大
pmap -x 1234 | sort -k3 -n -r | head -10

为什么 RSS 会被共享

动态库
libc、libssl 这些库在多个进程间共享同一份物理页。10 个进程用 libc，物理内存只有一份，但每个进程的 RSS 都算了进去。所以所有进程 RSS 之和会大于物理内存
fork 的 COW
fork 之后父子进程共享全部内存页，只在写入时才复制。Redis 的 bgsave、Nginx 的 worker 都靠这个省内存
共享内存 / tmpfs
shmget、mmap(MAP_SHARED)、tmpfs 文件，多个进程映射同一块页。注意 tmpfs 还占内存配额
KSM
Kernel Samepage Merging，内核主动扫描并合并内容相同的页。虚拟机场景常用，但会增加 CPU 开销
# 一个反直觉的验证：所有进程 RSS 之和 > 物理内存
$ ps -eo rss= | awk '{s+=$1} END {print s/1024/1024 " GB"}'
12.4 GB
$ free -g | awk '/Mem:/{print $2 " GB total"}'
16 GB
# 这不是内存泄漏，是共享页被重复计算

# 用 PSS 才是准的
$ sudo smem -t -k
# 或者遍历 /proc/*/smaps_rollup 累加 Pss
排查内存问题的正确顺序：
① 看趋势：RSS 是锯齿、阶梯还是线性？决定是正常、泄漏还是碎片。
② 看构成：RssAnon 涨 = 堆/栈（业务代码）；RssFile 涨 = 文件映射（可能是 mmap 了大文件）；RssShmem 涨 = 共享内存。
③ 看 PSS：确认到底占了多少真实内存，别被共享页误导。
④ 看 cgroup：memory.current 和 memory.events，确认是不是被容器限制卡住。
⑤ 最后才看代码：用 valgrind / pprof / 堆 dump 定位到具体分配点。
常见坑

十个踩过的坑

1
看 free 列的数值判断内存够不够
Linux 会主动用空闲内存做页缓存，free 列几乎总是很小。判断标准是 available 列，它才是"还能给新程序用多少"的估算值。
2
把 VSZ 当成实际内存占用
VSZ 是虚拟地址空间，包含大量从未访问的预留区域，动辄几十 GB。JVM 启动后 VSZ 可能有 8 GB 而 RSS 只有 500 MB。看 RSS 或 PSS，不看 VSZ。
3
把"所有进程 RSS 之和超过物理内存"当成异常
共享库、COW、共享内存都被重复计算了。要准确评估用 PSS（按共享进程数均摊），或者直接看 cgroup 的 memory.current。
4
把内存泄漏和内存溢出混为一谈
泄漏是"该释放没释放"，表现为缓慢上涨；溢出是"超过上限"，表现为突然失败。前者要查分配点，后者要查峰值和限制配置，排查路径完全不同。
5
认为"有 GC 就不会泄漏"
GC 只回收不可达对象。缓存无上限、集合只增不减、ThreadLocal 不 remove、监听器不注销，都会导致对象一直可达 —— 逻辑泄漏，GC 无能为力。
6
容器里 JVM 用固定 -Xmx
容器限制一变就 OOM 或浪费。更危险的是 -Xmx 等于容器上限 —— 堆之外的 Metaspace、Code Cache、线程栈、Direct Buffer 会直接撑爆。用 -XX:MaxRAMPercentage=70 并显式设 MaxDirectMemorySize。
7
用 swap 来"解决"内存不足
swap 的延迟是内存的几千倍（微秒级 vs 纳秒级）。si/so 持续非 0 时，服务延迟会断崖式恶化。swap 只该作为防止 OOM 的缓冲，不该当作扩容手段。
8
看到 RSS 不降就认定是泄漏
分配器（ptmalloc）free 后只是还给空闲链表，不一定还给内核。brk 只能从堆顶收缩，堆顶卡住就永远降不下来。触发一次 Full GC 后仍不降，且 RssAnon 平稳，多半是碎片。
9
在容器里用 free / nproc 判断资源
容器看到的是宿主机资源。内存限制看 /sys/fs/cgroup/memory.max，当前用量看 memory.current，OOM 证据看 memory.events 的 oom_kill 计数。
10
线程栈开太大导致线程创建失败
默认 8 MB 栈 × 1000 线程 = 8 GB 虚拟空间。虽然物理内存不一定用到，但会撞上 max_map_count 或地址空间上限。线程多时把栈调到 256 KB~1 MB。
自测

六道题

1. 某进程 VSZ 显示 8 GB，RSS 显示 500 MB。说明？
正确答案：只访问了 500 MB。虚拟内存允许"承诺"远大于实际使用，缺页时才分配物理页。JVM 启动时就会预留大块虚拟地址。判断实际占用要看 RSS 或 PSS，不是 VSZ。
2. 容器限制 4 GB，JVM 应该怎么配堆？
正确答案：MaxRAMPercentage 70 左右。堆之外还有 Metaspace、Code Cache（默认 240 MB）、线程栈、Direct ByteBuffer、GC 元数据。-Xmx 等于容器上限必然 OOMKilled；-Xmx512m 又浪费资源。还要显式设 MaxDirectMemorySize。
3. free 输出里 free 列只有 1 GB，available 列有 9 GB，说明？
正确答案：页缓存可回收。Linux 主动用空闲内存做文件缓存，free 列小是设计使然。available 才是"还能给新程序用多少"的估算，9 GB 说明内存充足。只有 available 低于 total 的 10% 才需要警惕。
4. 服务 RSS 三天内阶梯式上涨（涨一段平一段再涨），触发 Full GC 后不回落。最可能是？
正确答案：内存泄漏。阶梯式上涨说明每次触发某条业务路径就漏一点；Full GC 后不回落说明这些对象是可达的，GC 无能为力。碎片通常表现为"涨到某个水位后平稳"，不会持续阶梯上涨。下一步用堆 dump 或 pprof 找分配点。
5. 三色标记法中，如果不加写屏障，并发标记可能出错的原因是？
正确答案：对象消失问题。需要同时满足两个条件：黑色对象插入指向白色对象的新边，且所有从灰色到该白色的路径都被删除。写屏障只要破坏其中任意一个就安全 —— 增量更新破坏前者（把黑色变回灰色），SATB 破坏后者（记录被删除的旧引用）。
6. 容器退出码 137，最可能的原因是？
正确答案：被 SIGKILL 杀。137 = 128 + 9，即收到信号 9。最常见的是 cgroup 内存超限。用 cat /sys/fs/cgroup/memory.events 看 oom_kill 计数，用 dmesg | grep -i "killed process" 看内核日志确认。

四个内存指标的区别

指标含义用来看什么
VSZ虚拟地址空间大小地址空间是否耗尽、mmap 是否过多。不代表实际占用
RSS常驻物理内存（含共享）趋势判断：锯齿/阶梯/线性
PSS共享页按进程数均摊准确评估真实占用，各进程 PSS 之和 ≈ 物理内存
USS进程独占的物理内存杀进程能释放多少，泄漏排查的首选

# 内核

vm.overcommit_ratio         # 模式 2 下的物理内存比例，默认 50

# glibc malloc
M_MMAP_THRESHOLD            # 默认 128 KB，超过走 mmap
M_TRIM_THRESHOLD            # 默认 128 KB，堆顶空闲超过就收缩

# jemalloc
dirty_decay_ms              # 默认 10000，脏页多久归还
muzzy_decay_ms              # 默认 0

# JVM
-XX:MaxRAMPercentage=70     # 按容器内存算堆上限
-XX:InitialRAMPercentage=70 # 初始堆，避免运行期扩容

术语表

术语一句话解释
虚拟内存每个进程独享的地址空间抽象，由页表映射到物理内存
overcommit允许分配的虚拟内存超过物理内存总量，靠缺页时再兑现
页表 page table虚拟页到物理页帧的映射表，x86-64 默认四级
PTE页表项，包含物理页帧号和各种标志位（Present/Dirty/Accessed/NX）
TLB地址翻译缓存，miss 时要走多级页表，代价等同一次 cache miss
大页 huge page2 MB 或 1 GB 的页，大幅提升 TLB 覆盖范围，代价是内部碎片
缺页中断 page fault访问的页没有有效映射时触发。minor 在内存里，major 要读磁盘
COW写时复制，fork 后共享页，写入时才真正复制
brk通过移动堆顶指针扩展堆，只能从顶部收缩
mmap建立独立的内存映射，free 时可 munmap 完整归还
RSS常驻物理内存，含与其他进程共享的页
PSS共享页按进程数均摊后的内存占用，最接近真实值
USS进程独占的物理内存，杀进程能释放的量
内部碎片分配出去的块比请求的大，多出的部分浪费在块内
外部碎片空闲总量够但分散，凑不出连续大块
三色标记GC 用白/灰/黑表示扫描进度，并发标记的正确性基础
写屏障 write barrier赋值时插入的钩子，记录引用变化，保证并发标记不漏对象
SATB按 GC 开始时的快照标记，删除引用时记录旧值
STWStop The World，GC 暂停所有业务线程的阶段
OOM Killer内存耗尽时内核按 badness 打分杀进程的机制
cgroup内核的资源隔离机制，memory.max 是容器的内存硬限制
